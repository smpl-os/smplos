# smplOS Shared AUR Packages
# These packages are built from the Arch User Repository
# Custom packages (src/shared/pkgbuilds/*) are also listed here.

# AUR helper (so users can install AUR packages after install)
paru-bin

# EWW - Universal widget system (works on Wayland and X11)
# smplOS fork with Wayland button-grab fixes (see pkgbuilds/eww-smplos).
# Upstream AUR `eww` (tag v0.6.0) has a bug where clicking a bar button
# no-ops until the mouse is moved. eww-smplos is a local PKGBUILD that
# builds from github.com/smpl-os/eww at a pinned commit.
eww-smplos

# Screenshot annotation tool (pre-compiled binary)
satty-bin

# Brave browser (pre-compiled binary from AUR)
brave-bin

# smplOS custom Nemo (built from src/shared/pkgbuilds/nemo-smpl)
nemo-smpl

# Grafium — knowledge graph & note-taking GUI (repackaged upstream .deb,
# built from src/shared/pkgbuilds/grafium-bin; not in the AUR yet)
grafium-bin

# Auto-snapshot before every pacman update (pacman hook for timeshift)
# Built from src/shared/pkgbuilds/timeshift-autosnap (git source, avoids
# Codeberg tarball endpoint which returns intermittent HTTP 500 errors)
# timeshift-autosnap

# Local AI dictation daemon (Whisper-based, fully offline speech-to-text)
voxtype-bin

# Mullvad VPN client (built from src/shared/pkgbuilds/mullvad-vpn-bin)
# Repackaged from the upstream .deb on mullvad.net and verified against
# Mullvad's PGP code-signing key -- avoids AUR maintainer trust chain.
# Daemon is enabled at boot; kill-switch service is opt-in (see .install).
mullvad-vpn-bin

# WCH ISP flasher for Settings > Keypad > Firmware (CH552 macro keypads).
# Only `keypad-ctl firmware flash` drives it (info + flash, never config).
wchisp

# Macro keypad daemon (src/shared/pkgbuilds/control-surface). Parked by its
# PENDING file until the source repository is published; see KEYPAD.md.
# control-surface

# Hyprland plugins/extras (optional)
# hyprland-plugins-git
